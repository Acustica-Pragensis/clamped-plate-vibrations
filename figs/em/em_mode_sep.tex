% Colour-blind-friendly palette (Okabe-Ito) with strong contrast on white.

\definecolor{lineorange}{HTML}{E69F00}
\definecolor{linegreen}{HTML}{009E73}

% Larger tick/axis text so the figure survives scaling to a journal column.
\pgfplotsset{
    every axis/.append style={
        tick label style={font=\large},
        label style={font=\Large},
        axis line style={line width=0.8pt},
        tick style={line width=0.8pt},
        legend style={
            font=\large,
            draw=gray!35,
            fill=white,
            fill opacity=0.90,
            text opacity=1,
            rounded corners=1pt,
            inner sep=3pt,
        },
    },
}
